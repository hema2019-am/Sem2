5A. Evaluating Feed Forward Deep Network for Regression using K-Fold Cross Validation
import pandas as pd
from keras.models import Sequential
from keras.layers import Dense
from scikeras.wrappers import KerasRegressor
from sklearn.model_selection import cross_val_score
from sklearn.model_selection import KFold
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import Pipeline

# Load Dataset
dataframe = pd.read_csv("housing.csv", delim_whitespace=True, header=None)

dataset = dataframe.values
X = dataset[:, 0:13]
Y = dataset[:, 13]

# Create Model
def wider_model():
    model = Sequential()

    model.add(Dense(15,
                    input_dim=13,
                    activation='relu'))

    model.add(Dense(13,
                    activation='relu'))

    model.add(Dense(1))

    model.compile(
        loss='mean_squared_error',
        optimizer='adam'
    )

    return model

# Pipeline
estimators = []
estimators.append(('standardize', StandardScaler()))
estimators.append(
    ('mlp',
     KerasRegressor(
         model=wider_model,
         epochs=100,
         batch_size=5,
         verbose=0
     ))
)

pipeline = Pipeline(estimators)

# K-Fold Validation
kfold = KFold(
    n_splits=10,
    shuffle=True,
    random_state=1
)

results = cross_val_score(
    pipeline,
    X,
    Y,
    cv=kfold
)

print("Mean Score:", results.mean())
print("Standard Deviation:", results.std())

What is K-Fold Cross Validation?
K-Fold Cross Validation divides the dataset into K equal parts and evaluates the model K times using different test folds.

Why use StandardScaler?
It standardizes features so that all inputs have similar scales.

Why use Regression?
Regression predicts continuous values such as house prices.

What is MSE?
Mean Squared Error measures the average squared difference between actual and predicted values.

Result

A Feed Forward Deep Neural Network was successfully evaluated for a regression task using 10-Fold Cross Validation, and the average model performance was measured using regression metrics.

6.  implementing regularization to avoid overfitting in binary
classification.
from matplotlib import pyplot
from sklearn.datasets import make_moons
from keras.models import Sequential
from keras.layers import Dense
from keras.regularizers import L1L2

# Generate Dataset
X, Y = make_moons(
    n_samples=100,
    noise=0.2,
    random_state=1
)

# Split Dataset
n_train = 30
trainX, testX = X[:n_train, :], X[n_train:, :]
trainY, testY = Y[:n_train], Y[n_train:]

# Create Model
model = Sequential()

model.add(Dense(
    500,
    input_dim=2,
    activation='relu',
    kernel_regularizer=L1L2(l1=0.001, l2=0.001)
))

model.add(Dense(
    1,
    activation='sigmoid'
))

# Compile Model
model.compile(
    loss='binary_crossentropy',
    optimizer='adam',
    metrics=['accuracy']
)

# Train Model
history = model.fit(
    trainX,
    trainY,
    validation_data=(testX, testY),
    epochs=100,
    verbose=1
)

# Plot Accuracy
pyplot.plot(history.history['accuracy'], label='Train')
pyplot.plot(history.history['val_accuracy'], label='Test')

pyplot.xlabel("Epochs")
pyplot.ylabel("Accuracy")
pyplot.legend()
pyplot.show()


VWhat is L1 Regularization?

Adds the absolute value of weights to the loss function.
Can reduce some weights to zero.

What is L2 Regularization?

Adds the squared value of weights to the loss function.
Prevents weights from becoming too large.

Why use both L1 and L2?

L1 performs feature selection.
L2 improves model stability.
Together they help reduce overfitting and improve generalization.

X = []

    X.append(data[i:i+3])
    Y.append(data[i+3])

X = np.array(X)

model = Sequential()

predicted_prices = scaler.inverse_transform(prediction)

8. Performing encoding and decoding of images using deep autoencoder
import numpy as np
import matplotlib.pyplot as plt
from keras.datasets import mnist
from keras.models import Model
from keras.layers import Input, Dense

# Load MNIST Dataset
(x_train, _), (x_test, _) = mnist.load_data()

# Use a smaller dataset for faster execution
x_train = x_train[:2000]
x_test = x_test[:5]

# Normalize images
x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

# Flatten images
x_train = x_train.reshape((2000, 784))
x_test = x_test.reshape((5, 784))

# ---------------- Encoder ----------------
input_img = Input(shape=(784,))
encoded = Dense(64, activation="relu")(input_img)

# ---------------- Decoder ----------------
decoded = Dense(784, activation="sigmoid")(encoded)

# Autoencoder Model
autoencoder = Model(input_img, decoded)

# Encoder Model
encoder = Model(input_img, encoded)

# Decoder Model
encoded_input = Input(shape=(64,))
decoder_layer = autoencoder.layers[-1]
decoder = Model(encoded_input, decoder_layer(encoded_input))

# Compile
autoencoder.compile(
    optimizer="adam",
    loss="binary_crossentropy"
)

# Train
autoencoder.fit(
    x_train,
    x_train,
    epochs=5,
    batch_size=64,
    shuffle=True,
    verbose=1
)

# Encode Images
encoded_imgs = encoder.predict(x_test, verbose=0)

# Decode Images
decoded_imgs = decoder.predict(encoded_imgs, verbose=0)

# Display Results
plt.figure(figsize=(12,6))

for i in range(5):

    # Original Image
    plt.subplot(3,5,i+1)
    plt.imshow(x_test[i].reshape(28,28), cmap='gray')
    plt.title("Original")
    plt.axis("off")

    # Encoded Image
    plt.subplot(3,5,i+6)
    plt.imshow(encoded_imgs[i].reshape(8,8), cmap='gray')
    plt.title("Encoded")
    plt.axis("off")

    # Decoded Image
    plt.subplot(3,5,i+11)
    plt.imshow(decoded_imgs[i].reshape(28,28), cmap='gray')
    plt.title("Decoded")
    plt.axis("off")

plt.tight_layout()
plt.show()

Viva Questions

What is an Autoencoder?
An autoencoder is a neural network that learns to compress (encode) input data and then reconstruct (decode) it with minimal loss of information.

What is Encoding?
Encoding is the process of converting input data into a lower-dimensional representation.

What is Decoding?
Decoding reconstructs the original data from the encoded representation.

Why is the decoded image sometimes blurry?
The decoded image may appear blurry because some information is lost during compression, especially when the encoded representation is much smaller than the original image or when the model has not been trained sufficiently.

Applications of Autoencoders

Image Compression
Image Denoising
Feature Extraction
Dimensionality Reduction
Anomaly Detection

10. Aim: Denoising of images using autoencoder. 
import keras
from keras.datasets import mnist
from keras import layers
import numpy as np
import matplotlib.pyplot as plt

# Load MNIST Dataset
(X_train, _), (X_test, _) = mnist.load_data()

# Normalize Images
X_train = X_train.astype('float32') / 255.
X_test = X_test.astype('float32') / 255.

# Reshape Images
X_train = X_train.reshape((len(X_train), 28, 28, 1))
X_test = X_test.reshape((len(X_test), 28, 28, 1))

# Add Gaussian Noise
noise_factor = 0.5

X_train_noisy = X_train + noise_factor * np.random.normal(
    loc=0.0,
    scale=1.0,
    size=X_train.shape
)

X_test_noisy = X_test + noise_factor * np.random.normal(
    loc=0.0,
    scale=1.0,
    size=X_test.shape
)

# Clip values between 0 and 1
X_train_noisy = np.clip(X_train_noisy, 0., 1.)
X_test_noisy = np.clip(X_test_noisy, 0., 1.)

# Build Autoencoder
input_img = keras.Input(shape=(28, 28, 1))

# Encoder
x = layers.Conv2D(32, (3,3), activation='relu', padding='same')(input_img)
x = layers.MaxPooling2D((2,2), padding='same')(x)

x = layers.Conv2D(32, (3,3), activation='relu', padding='same')(x)
encoded = layers.MaxPooling2D((2,2), padding='same')(x)

# Decoder
x = layers.Conv2D(32, (3,3), activation='relu', padding='same')(encoded)
x = layers.UpSampling2D((2,2))(x)

x = layers.Conv2D(32, (3,3), activation='relu', padding='same')(x)
x = layers.UpSampling2D((2,2))(x)

decoded = layers.Conv2D(
    1,
    (3,3),
    activation='sigmoid',
    padding='same'
)(x)

# Create Model
autoencoder = keras.Model(input_img, decoded)

# Compile Model
autoencoder.compile(
    optimizer='adam',
    loss='binary_crossentropy'
)

# Train Model
autoencoder.fit(
    X_train_noisy,
    X_train,
    epochs=3,
    batch_size=128,
    shuffle=True,
    validation_data=(X_test_noisy, X_test),
    verbose=1
)

# Predict Denoised Images
predictions = autoencoder.predict(X_test_noisy)

# -------------------- Display Results --------------------

n = 10

fig, axes = plt.subplots(3, n, figsize=(20,6))

for i in range(n):

    # Original Images
    axes[0, i].imshow(X_test[i].reshape(28,28), cmap='gray')
    axes[0, i].axis('off')

    # Noisy Images
    axes[1, i].imshow(X_test_noisy[i].reshape(28,28), cmap='gray')
    axes[1, i].axis('off')

    # Denoised Images
    axes[2, i].imshow(predictions[i].reshape(28,28), cmap='gray')
    axes[2, i].axis('off')

# Row Labels
axes[0,0].set_ylabel("Original", fontsize=14)
axes[1,0].set_ylabel("Noisy", fontsize=14)
axes[2,0].set_ylabel("Denoised", fontsize=14)

plt.suptitle("Denoising of Images using Autoencoder", fontsize=18)

plt.tight_layout()
plt.show()
