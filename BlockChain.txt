1A. Develop a secure messaging application where users can exchange
messages securely using RSA encryption. Implement a mechanism
for generating RSA key pairs and encrypting/decrypting messages.

RSA is an asymmetric encryption algorithm.
Each user has a public key and a private key.
The sender encrypts the message using the receiver's public key.
Only the receiver can decrypt the message using their private key.
This ensures secure communication over an insecure network.

from Crypto.PublicKey import RSA
from Crypto.Cipher import PKCS1_OAEP

# User class
class User:
    def __init__(self, name):
        self.name = name
        self.key = RSA.generate(2048)
        self.private_key = self.key
        self.public_key = self.key.publickey()

    def encrypt_message(self, message, receiver_public_key):
        cipher = PKCS1_OAEP.new(receiver_public_key)
        return cipher.encrypt(message.encode())

    def decrypt_message(self, encrypted_message):
        cipher = PKCS1_OAEP.new(self.private_key)
        return cipher.decrypt(encrypted_message).decode()

# Create two users
user1 = User("Alice")
user2 = User("Bob")

# Alice sends a message to Bob
message = "Hello Bob! This is a secure message."
print("Original Message:", message)

encrypted_msg = user1.encrypt_message(message, user2.public_key)
print("\nEncrypted Message:")
print(encrypted_msg)

# Bob decrypts the message
decrypted_msg = user2.decrypt_message(encrypted_msg)
print("\nDecrypted Message:")
print(decrypted_msg)

import binascii
import datetime
import collections
from Crypto.PublicKey import RSA
from Crypto import Random
from Crypto.Hash import SHA
from Crypto.Signature import PKCS1_v1_5

class Client:
    def __init__(self):
        random_gen = Random.new().read
        self._private_key = RSA.generate(1024, random_gen)
        self._public_key = self._private_key.publickey()

    @property
    def identity(self):
        return binascii.hexlify(
            self._public_key.exportKey(format='DER')
        ).decode('ascii')

class Transaction:
    def __init__(self, sender, recipient, value):
        self.sender = sender
        self.recipient = recipient
        self.value = value
        self.time = datetime.datetime.now()

    def to_dict(self):
        return collections.OrderedDict({
            'sender': self.sender.identity,
            'recipient': self.recipient,
            'value': self.value,
            'time': self.time
        })

    def sign_transaction(self):
        signer = PKCS1_v1_5.new(self.sender._private_key)
        h = SHA.new(str(self.to_dict()).encode('utf8'))
        return binascii.hexlify(signer.sign(h)).decode('ascii')

def display_transactions(transactions):
    print("\n========== TRANSACTION LIST ==========\n")

    for i, transaction in enumerate(transactions, start=1):
        data = transaction.to_dict()

        print(f"Transaction #{i}")
        print("Sender    :", data['sender'][:30] + "...")
        print("Recipient :", data['recipient'][:30] + "...")
        print("Amount    :", data['value'])
        print("Time      :", data['time'])
        print("-" * 50)

class Account:
    def __init__(self, name, balance):
        self.name = name
        self.balance = balance

class Transaction:
    def __init__(self, sender, receiver, amount):
        self.sender = sender
        self.receiver = receiver
        self.amount = amount

    def transfer(self):
        if self.sender.balance >= self.amount:
            self.sender.balance -= self.amount
            self.receiver.balance += self.amount

            print("Transaction Successful!")
            print(f"{self.amount} transferred from {self.sender.name} to {self.receiver.name}")
        else:
            print("Insufficient Balance!")

2A. Write a python program to demonstrate mining.
Mining is the process of finding a nonce value such 
that the generated hash satisfies a specific condition (difficulty level).
Steps:
Take a message/block data.
Generate a hash using SHA-256.
Change the nonce repeatedly.
Check if the hash starts with the required prefix.
If the condition is satisfied, the block is considered mined.

import hashlib
def sha256(message):
    return hashlib.sha256(message.encode('ascii')).hexdigest()

def mine(message, difficulty=2):
    prefix = '1' * difficulty
    print("Prefix:", prefix)

    for i in range(100):   # Only 100 iterations
        digest = sha256(str(hash(message)) + str(i))
        print("Testing =>", digest)

        if digest.startswith(prefix):
            print("\nNonce Found:", i)
            print("Hash:", digest)
            return i

    print("\nNo valid nonce found within 100 iterations.")
    return None

n = mine("test message", 2)
print("Nonce Value:", n)

Function Overloading
Multiple functions with the same name but different parameters.
Example: add(uint,uint) and add(uint,uint,uint).
Mathematical Functions
Perform arithmetic operations such as maximum and square calculations.
Cryptographic Functions
keccak256() – Ethereum's primary hashing algorithm.
sha256() – Secure Hash Algorithm 256-bit.
ripemd160() – Produces a 160-bit cryptographic hash.

pragma solidity ^0.8.0;

    // Function Overloading
    function add(uint a, uint b) public pure returns(uint) {
        return a + b;
    }

    function add(uint a, uint b, uint c) public pure returns(uint) {
        return a + b + c;
    }

    // Mathematical Functions
    function getMax(uint a, uint b) public pure returns(uint) {
        if(a > b)
            return a;

interface IDisplay {
    function display() external pure returns (string memory);
}

// Abstract Contract
abstract contract Person {
    string public name;

    constructor(string memory _name) {
        name = _name;
    }

    function getRole() public view virtual returns (string memory);
}

// Child Contract
contract Student is Person, IDisplay {

    uint public rollNo;

    constructor(string memory _name, uint _rollNo)

    {
        rollNo = _rollNo;

        public
        pure
