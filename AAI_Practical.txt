This code:
Downloads handwritten digit images
Trains a deep learning model to recognize them
Tests how accurate it is
Predicts digits on new images

import tensorflow as tf
from tensorflow.keras import layers, models
import numpy as np
import matplotlib.pyplot as plt

# Load MNIST
mnist = tf.keras.datasets.mnist
(x_train, y_train), (x_test, y_test) = mnist.load_data()

# Normalize
x_train = x_train / 255.0
x_test = x_test / 255.0

# Add channel dimension
x_train = x_train[..., tf.newaxis]
x_test = x_test[..., tf.newaxis]

# Build model
model = models.Sequential([
    layers.Input(shape=(28, 28, 1)),
    layers.Conv2D(32, (3, 3), activation='relu'),
    layers.MaxPooling2D((2, 2)),
    
    layers.Conv2D(64, (3, 3), activation='relu'),
    layers.MaxPooling2D((2, 2)),
    
    layers.Flatten(),
    layers.Dense(64, activation='relu'),
    layers.Dense(10, activation='softmax')
])

# Compile
model.compile(
    optimizer='adam',
    loss='sparse_categorical_crossentropy',
    metrics=['accuracy']
)

# Train
model.fit(x_train, y_train, epochs=5, validation_split=0.1)

# Evaluate
test_loss, test_acc = model.evaluate(x_test, y_test)
print("Test accuracy:", test_acc)

# Predict
predictions = model.predict(x_test, verbose=0)

# Plot predictions
for i in range(5):
    plt.imshow(x_test[i].squeeze(), cmap='gray')
    plt.title(f"Predicted: {np.argmax(predictions[i])} / Actual: {y_test[i]}")
    plt.axis('off')
    plt.show()

This is a basic Sentiment Analysis program using a tiny neural network for text.
The code trains a neural network to classify short text sentences as Positive 
or Negative sentiment, then uses it to predict sentiment of new sentences.

import tensorflow as tf
from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Embedding, Flatten, Dense
import numpy as np

# -----------------------------
# Training Data
# -----------------------------
texts = [
    "I love this product",
    "This is the worst movie",
    "I am so happy",
    "I hate this book",
    "What a great day",
    "I am very disappointed"
]

labels = [1, 0, 1, 0, 1, 0]  # 1 = Positive, 0 = Negative

# -----------------------------
# Tokenizer
# -----------------------------
tokenizer = Tokenizer(oov_token="<OOV>")
tokenizer.fit_on_texts(texts)

# Convert to sequences
sequences = tokenizer.texts_to_sequences(texts)
padded = pad_sequences(sequences, padding='post', maxlen=5)

# Vocabulary size
vocab_size = len(tokenizer.word_index) + 1

# -----------------------------

model = Sequential([

    Dense(6, activation='relu'),

# -----------------------------

    loss='binary_crossentropy',
    optimizer='adam',

# -----------------------------

3. Building a natural language processing (NLP) model for sentiment analysis or text classification
What This Code Does (Simple)

Trains a neural network on IMDB movie reviews
Learns word embeddings
Classifies reviews as Positive / Negative
Takes new raw English sentences
Converts them to IMDB word indices
Predicts sentiment

import tensorflow as tf
from tensorflow.keras.datasets import imdb
from tensorflow.keras.preprocessing.sequence import pad_sequences
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Embedding, GlobalAveragePooling1D, Dense
import re

# -----------------------------
# 1. Load IMDB Dataset
# -----------------------------
num_words = 10000
maxlen = 256

(x_train, y_train), (x_test, y_test) = imdb.load_data(num_words=num_words)

# Pad sequences
x_train = pad_sequences(x_train, maxlen=maxlen)
x_test  = pad_sequences(x_test, maxlen=maxlen)

# -----------------------------
# 2. Build Model
# -----------------------------
model = Sequential([
    Embedding(input_dim=num_words, output_dim=16),
    GlobalAveragePooling1D(),
    Dense(16, activation='relu'),
    Dense(1, activation='sigmoid')
])

# -----------------------------
# 3. Compile
# -----------------------------
model.compile(
    optimizer='adam',
    loss='binary_crossentropy',
    metrics=['accuracy']
)

# -----------------------------
# 4. Train
# -----------------------------
model.fit(
    x_train,
    y_train,
    epochs=10,
    batch_size=512,
    validation_split=0.2,
    verbose=2
)

# -----------------------------
# 5. Evaluate
# -----------------------------
loss, accuracy = model.evaluate(x_test, y_test, verbose=2)
print("Loss:", loss)
print("Accuracy:", accuracy)

# -----------------------------
# 6. Predict New Reviews
# -----------------------------
word_index = imdb.get_word_index()

reviews = [
    "This movie was amazing! I liked and enjoyed every moment of it",
    "This movie was terrible. The acting was awful and the plot was worst."
]

encoded_reviews = []

for review in reviews:
    # Clean text
    cleaned = re.sub(r"[^\w\s]", "", review.lower())
    words = cleaned.split()

    # Encode words (use 2 for unknown, add offset of 3)
    encoded = [word_index.get(word, 2) + 3 for word in words]
    encoded_reviews.append(encoded)

# Pad new reviews
padded_reviews = pad_sequences(encoded_reviews, maxlen=maxlen)

# Predict
predictions = model.predict(padded_reviews, verbose=0).flatten()

# Convert to labels
output_labels = ["Positive" if score >= 0.5 else "Negative" for score in predictions]

# -----------------------------
# 7. Output Results
# -----------------------------
for review, score, label in zip(reviews, predictions, output_labels):
    print("\nReview:", review)
    print(f"Sentiment score: {score:.4f}")
    print("Predicted sentiment:", label)

4. Creating a chatbot using advanced techniques like transformer model
This builds a BERT-based Question Answering chatbot that:

Reads a paragraph (context)
Takes natural language questions
Uses BERT to extract answers from context
Returns the best answer span

This is Transformer-based NLP, much more advanced than your earlier models.
import tensorflow as tf
import transformers
import warnings

from transformers import AutoTokenizer, TFBertForQuestionAnswering, pipeline

print("Transformers version:", transformers.__version__)
print("TensorFlow version:", tf.__version__)
print("Model path or name:", "deepset/bert-base-cased-squad2")

# -----------------------------
context = """
The national animal of India is the Bengal tiger.
The national bird of India is the Indian peacock.
The largest terrestrial mammal in India is the Indian elephant.
The banyan tree is known for producing aerial roots.
The mango is believed to have originated in northeast India.
The symbol of longevity in many cultures is the tortoise.
"""

# Questions
# -----------------------------

    "What is the National animal of India?",
    "What is the National bird of India?",
    "What is the largest terrestrial mammal in India?",
    "Which tree produces aerial roots?",
    "Which fruit originated in northeast India?",
    "What is the symbol of longevity?"
]

# Load Tokenizer & Model
# -----------------------------

model = TFBertForQuestionAnswering.from_pretrained(
    "deepset/bert-base-cased-squad2",
    from_pt=True
)

# -----------------------------
# Build QA Pipeline
# -----------------------------
nlp = pipeline(
    "question-answering",
    model=model,
    tokenizer=tokenizer
)

# -----------------------------
# Chatbot Function (FIXED)
# -----------------------------
def chatbot(question, context):
    output = nlp({
        "question": question,
        "context": context
    })
    return output["answer"]

# -----------------------------
# Ask Questions
# -----------------------------
for q in questions:
    print("\nQuestion:", q)
    print("Answer:", chatbot(q, context))
5. Developing a recommendation system using collaborative filtering or deep learning approaches
This code trains a neural network to learn user and item preferences using embeddings and then predicts how much a user will like each item.

This is called Neural Collaborative Filtering.
import tensorflow as tf
import numpy as np

# Example user-item interactions
# user_id, item_id, rating
data = np.array([
    [0, 0, 5],
    [0, 1, 4],
    [1, 0, 4],
    [1, 3, 1],
    [2, 0, 1],
    [2, 3, 5],
    [3, 2, 5],
    [3, 3, 4],
])

users = data[:, 0]
items = data[:, 1]
ratings = data[:, 2]

num_users = 4
num_items = 4

# User & Item Embeddings
user_input = tf.keras.Input(shape=(1,))
item_input = tf.keras.Input(shape=(1,))

user_emb = tf.keras.layers.Embedding(num_users, 8)(user_input)
item_emb = tf.keras.layers.Embedding(num_items, 8)(item_input)

user_vec = tf.keras.layers.Flatten()(user_emb)
item_vec = tf.keras.layers.Flatten()(item_emb)

# Dot product (similarity)
dot = tf.keras.layers.Dot(axes=1)([user_vec, item_vec])

model = tf.keras.Model([user_input, item_input], dot)

model.compile(optimizer='adam', loss='mse')

# Train
model.fit([users, items], ratings, epochs=50, verbose=0)

# Recommend for User 0
user_id = 0

print("\nRecommendations for User 1:")

for item_id in range(num_items):
    pred_rating = model.predict([np.array([user_id]), np.array([item_id])], verbose=0)
    print(f"Item {item_id+1} → Predicted Rating: {pred_rating[0][0]:.2f}")

6. Developing a recommendation system using collaborative filtering or deep learning approaches. Using movielens dataset
To run 
first>  set TF_USE_LEGACY_KERAS=1

This code trains a deep learning collaborative filtering model on the MovieLens dataset to recommend movies to users based on learned user and movie embeddings.

This is similar in concept to Netflix / Amazon / Spotify recommendations.


import numpy as np
import tensorflow as tf
import tensorflow_datasets as tfds
import tensorflow_recommenders as tfrs

# -----------------------------
# 1. Load MovieLens 100K Dataset
# -----------------------------
ratings = tfds.load("movielens/100k-ratings", split="train")
movies  = tfds.load("movielens/100k-movies", split="train")

# Keep only what we need
ratings = ratings.map(lambda x: {
    "user_id": x["user_id"],
    "movie_title": x["movie_title"],
})

movies = movies.map(lambda x: x["movie_title"])

# -----------------------------
# 2. Get Unique Users & Movies
# -----------------------------
user_ids = ratings.map(lambda x: x["user_id"])
movie_titles = ratings.map(lambda x: x["movie_title"])

unique_user_ids = np.unique(np.concatenate(list(user_ids.batch(1000))))
unique_movie_titles = np.unique(np.concatenate(list(movie_titles.batch(1000))))

# -----------------------------
# 3. Train / Test Split
# -----------------------------
tf.random.set_seed(42)
shuffled = ratings.shuffle(100_000, seed=42, reshuffle_each_iteration=False)

train = shuffled.take(80_000)
test  = shuffled.skip(80_000).take(20_000)

# -----------------------------
# 4. User & Movie Models (Embeddings)
# -----------------------------
embedding_dim = 32

user_model = tf.keras.Sequential([
    tf.keras.layers.StringLookup(vocabulary=unique_user_ids, mask_token=None),
    tf.keras.layers.Embedding(len(unique_user_ids) + 1, embedding_dim)
])

movie_model = tf.keras.Sequential([
    tf.keras.layers.StringLookup(vocabulary=unique_movie_titles, mask_token=None),
    tf.keras.layers.Embedding(len(unique_movie_titles) + 1, embedding_dim)
])

# -----------------------------
# 5. Retrieval Model (Collaborative Filtering)
# -----------------------------
class MovieLensModel(tfrs.Model):
    def __init__(self, user_model, movie_model):
        super().__init__()
        self.user_model = user_model
        self.movie_model = movie_model
        self.task = tfrs.tasks.Retrieval(
            metrics=tfrs.metrics.FactorizedTopK(
                candidates=movies.batch(128).map(movie_model)
            )
        )

    def compute_loss(self, features, training=False):
        user_embeddings = self.user_model(features["user_id"])
        movie_embeddings = self.movie_model(features["movie_title"])
        return self.task(user_embeddings, movie_embeddings)

# -----------------------------
# 6. Compile & Train
# -----------------------------
model = MovieLensModel(user_model, movie_model)
model.compile(optimizer=tf.keras.optimizers.Adagrad(0.1))

model.fit(train.batch(1024), epochs=3)

# -----------------------------
# 7. Build Recommendation Index
# -----------------------------
index = tfrs.layers.factorized_top_k.BruteForce(model.user_model)
index.index_from_dataset(
    movies.batch(100).map(lambda title: (title, movie_model(title)))
)

# -----------------------------
# 8. Recommend Movies for a User
# -----------------------------
user_id = unique_user_ids[0]

scores, titles = index(tf.constant([user_id]))

print(f"\nTop recommendations for user {user_id.decode('utf-8')}:\n")


for title in titles[0, :5]:
    print(title.numpy().decode("utf-8"))

6. Building a deep learning model for time series forecasting
This code trains a neural network to predict the next day’s sales using the previous 3 days of sales data.
This is time series forecasting using a sliding window + neural network.
The sliding window method is a technique to convert time series data into supervised 
learning data by using a fixed number of past values to predict the next value.
import numpy as np
import pandas as pd
import tensorflow as tf
from sklearn.preprocessing import MinMaxScaler
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import LSTM, Dense
import matplotlib.pyplot as plt

# -----------------------------
# 1. Load Dataset
# -----------------------------
url = "https://raw.githubusercontent.com/jbrownlee/Datasets/master/daily-min-temperatures.csv"
df = pd.read_csv(url)

# Use only temperature column
temps = df["Temp"].values.reshape(-1, 1)

# -----------------------------
# 2. Train / Test Split
# -----------------------------
train_size = int(len(temps) * 0.8)
train, test = temps[:train_size], temps[train_size:]

# -----------------------------
# 3. Scale Data
# -----------------------------
scaler = MinMaxScaler()
train_scaled = scaler.fit_transform(train)
test_scaled  = scaler.transform(test)

# -----------------------------
# 4. Sliding Window Function
# -----------------------------
def create_sequences(data, n_steps):
    X, y = [], []
    for i in range(len(data) - n_steps):
        X.append(data[i:i+n_steps])
        y.append(data[i+n_steps])
    return np.array(X), np.array(y)

n_steps = 30  # past 30 days

X_train, y_train = create_sequences(train_scaled, n_steps)
X_test,  y_test  = create_sequences(test_scaled, n_steps)

# -----------------------------
# 5. Build LSTM Model
# -----------------------------
model = Sequential([
    LSTM(50, activation="tanh", input_shape=(n_steps, 1)),
    Dense(1)
])

model.compile(optimizer="adam", loss="mse")

model.summary()

# -----------------------------
# 6. Train Model
# -----------------------------
history = model.fit(
    X_train, y_train,
    epochs=10,
    batch_size=32,
    validation_split=0.1,
    verbose=1
)

# -----------------------------
# 7. Evaluate
# -----------------------------
test_loss = model.evaluate(X_test, y_test, verbose=0)
print("Test MSE:", test_loss)

# -----------------------------
# 8. Predict Next Day Temperature
# -----------------------------
last_window = test_scaled[-n_steps:]
last_window = last_window.reshape(1, n_steps, 1)

next_temp_scaled = model.predict(last_window, verbose=0)
next_temp = scaler.inverse_transform(next_temp_scaled)

print(f"Predicted next day minimum temperature: {next_temp[0][0]:.2f} °C")

# -----------------------------
# 9. Plot Predictions vs Actual
# -----------------------------
train_preds = model.predict(X_train, verbose=0)
test_preds  = model.predict(X_test, verbose=0)

train_preds_inv = scaler.inverse_transform(train_preds)
y_train_inv     = scaler.inverse_transform(y_train)

test_preds_inv = scaler.inverse_transform(test_preds)
y_test_inv     = scaler.inverse_transform(y_test)

plt.figure(figsize=(10,5))
plt.plot(y_test_inv, label="Actual")
plt.plot(test_preds_inv, label="Predicted")
plt.title("Daily Minimum Temperature Forecast (Test Set)")
plt.xlabel("Days")
plt.ylabel("Temperature (°C)")
plt.legend()
plt.show()
